set(CARGO_BUILD_FLAG "")
set(CARGO_TARGET_SUBDIR "debug")

if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CARGO_BUILD_FLAG "--release")
    set(CARGO_TARGET_SUBDIR "release")
endif()

set(OUTPUT_DIR "${PROJECT_SOURCE_DIR}/out/bevy")
set(RUST_BIN_NAME "bevy-test${CMAKE_EXECUTABLE_SUFFIX}")
set(RUST_BUILD_PATH "${CMAKE_CURRENT_SOURCE_DIR}/target/${CARGO_TARGET_SUBDIR}/${RUST_BIN_NAME}")

add_custom_target(bevy-test ALL
    COMMAND cargo build ${CARGO_BUILD_FLAG}
    COMMAND ${CMAKE_COMMAND} -E make_directory "${OUTPUT_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RUST_BUILD_PATH}" "${OUTPUT_DIR}/${RUST_BIN_NAME}"
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    BYPRODUCTS "${OUTPUT_DIR}/${RUST_BIN_NAME}"
    USES_TERMINAL
    VERBATIM
)
